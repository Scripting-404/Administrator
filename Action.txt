repeat task.wait() until game:IsLoaded()

local TARGET_URL = "https://raw.githubusercontent.com/Scripting-404/Administrator/refs/heads/main/Target.txt"
local EXECUTE_URL = "https://raw.githubusercontent.com/Scripting-404/Administrator/refs/heads/main/Execute.txt"

local CHECK_INTERVAL = 1

local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local LocalUsername = LocalPlayer.Name

local lastTargetRaw
local lastExecuteRaw

local isRunning = true
local isExecuting = false

local function httpGet(url)
    local success, result = pcall(function()
        return game:HttpGet(url)
    end)

    if not success then
        warn("[Loader] HTTP Error:", result)
        return nil
    end

    return result
end

local function isUsernameWhitelisted(username, raw)
    if not raw then
        return false
    end

    username = username:lower()

    for line in raw:gmatch("[^\r\n]+") do
        line = line:match("^%s*(.-)%s*$")

        if line ~= "" and not line:match("^#") then
            local targetUsername = line:match("^[^|]+")

            if targetUsername then
                targetUsername = targetUsername:match("^%s*(.-)%s*$")

                if targetUsername:lower() == username then
                    return true
                end
            end
        end
    end

    return false
end

local function executeRaw(source)
    if not source or source == "" then
        warn("[Loader] Execute raw is empty.")
        return false
    end

    if isExecuting then
        return false
    end

    isExecuting = true

    local success, result = pcall(function()
        local func, compileError = loadstring(source)

        if not func then
            error("Compile error: " .. tostring(compileError))
        end

        func()
    end)

    isExecuting = false

    if not success then
        warn("[Loader] Execute error:", result)
        return false
    end

    print("[Loader] Execute successful.")
    return true
end

task.spawn(function()
    print("[Loader] Started.")
    print("[Loader] Username:", LocalUsername)

    while isRunning do
        local targetRaw = httpGet(TARGET_URL)

        if targetRaw then
            local targetChanged = targetRaw ~= lastTargetRaw
            lastTargetRaw = targetRaw

            if isUsernameWhitelisted(LocalUsername, targetRaw) then
                local executeSource = httpGet(EXECUTE_URL)

                if executeSource and executeSource ~= lastExecuteRaw then
                    print("[Loader] New execute raw found.")

                    if executeRaw(executeSource) then
                        lastExecuteRaw = executeSource
                    end
                elseif executeSource then
                    print("[Loader] Execute raw has not changed.")
                end
            elseif targetChanged then
                print("[Loader] Username is not in the whitelist.")
            end
        end

        task.wait(CHECK_INTERVAL)
    end
end)

getgenv().StopWhitelistLoader = function()
    isRunning = false
    print("[Loader] Stopped.")
end
